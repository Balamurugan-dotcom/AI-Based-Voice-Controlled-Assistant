PASS = " PASS"
FAIL = " FAIL"
results = []

# -------------------------------
# HELPERS
# -------------------------------

def wait(s):
    time.sleep(s)

def section(title):
    print(f"\n{'═'*55}")
    print(f"   {title}")
    print(f"{'═'*55}")

def log(name, status, note=""):
    tag = PASS if status else FAIL
    msg = f"   {tag}  {name}"
    if note:
        msg += f"  ({note})"
    print(msg)
    results.append((name, PASS if status else FAIL))

def safe(name, func, note=""):
    try:
        result = func()
        log(name, result is not False, note)
    except Exception as e:
        log(name, False, str(e))

def navigate_to(url):
    hotkey("ctrl", "l")
    wait(0.4)
    import pyperclip
    pyperclip.copy(url)
    hotkey("ctrl", "v")
    wait(0.3)
    press_key("enter")

def search_in_browser(query):
    hotkey("ctrl", "l")
    wait(0.4)
    import pyperclip
    pyperclip.copy(query)
    hotkey("ctrl", "v")
    wait(0.3)
    press_key("enter")

# TEST 8: OCR + BRAIN
# ══════════════════════════════════════════════════════
def test_ocr_and_brain():
    section("TEST 8: OCR + AI BRAIN")

    # OCR test
    try:
        from screen import capture_screen
        from ocr import extract_text

        open_app("notepad")
        wait(3)
        type_text("OCR Test: Hello World 12345")
        wait(0.5)

        img = capture_screen()
        t0 = time.time()
        ocr_data = extract_text(img)
        elapsed = time.time() - t0

        text_items = [x for x in ocr_data if x.get("source") != "icon"]
        found = any("ocr" in x["text"].lower() or "hello" in x["text"].lower() for x in text_items)

        print(f"\n    OCR Results:")
        print(f"      Items detected : {len(text_items)}")
        print(f"      Time taken     : {elapsed:.2f}s")
        print(f"      Found test text: {found}")

        log("OCR detects text",        len(text_items) > 10)
        log("OCR finds typed text",    found)
        log("OCR under 10s",           elapsed < 10)

        close_window()
        wait(1)
        press_key("n")
        wait(1)

    except Exception as e:
        log("OCR test", False, str(e))

    # Brain / LLM test
    try:
        from brain import call_llm, get_plan

        print("\n    Testing LLM connection...")
        response = call_llm("Reply with just the word: WORKING")
        log("LLM connection",  bool(response and len(response) > 0), response)

        print("\n    Testing AI planning...")
        plan = get_plan("open notepad and type hello world", [{"text": "Notepad"}])
        log("AI generates plan",        bool(plan and len(plan) > 0))
        log("Plan has correct actions", any(s.get("action") in ["open_app","type"] for s in plan) if plan else False)
        if plan:
            print(f"      Plan: {plan}")

    except Exception as e:
        log("Brain test", False, str(e))

def main():
    print("\n" + "█"*55)
    print("   DESKTOP AGENT — IN-DEPTH TEST SUITE")
    print("█"*55)
    print("\n  Tests: Browser, YouTube, Notepad, Calculator,")
    print("         File Explorer, Window Mgmt, System,")
    print("         OCR, AI Brain, End-to-End")
    print("\n    Keep hands off keyboard & mouse!")
    print("  Starting in 5 seconds...\n")
    time.sleep(5)

    # Run all test groups
    test_browser_google()
    test_youtube()
    test_notepad()
    test_calculator()
    test_file_explorer()
    test_window_management()
    test_system_controls()
    test_ocr_and_brain()
    test_end_to_end()

    # ══════════════════════════════
    # FINAL REPORT
    # ══════════════════════════════
    section("FINAL REPORT")

    passed  = [r for r in results if r[1] == PASS]
    failed  = [r for r in results if r[1] == FAIL]
    total   = len(results)
    pct     = int(len(passed) / total * 100) if total else 0

    print(f"\n  Total Tests : {total}")
    print(f"   Passed   : {len(passed)}  ({pct}%)")
    print(f"   Failed   : {len(failed)}")

    if failed:
        print(f"\n   FAILED TESTS:")
        for name, _ in failed:
            print(f"     • {name}")

    if pct == 100:
        print("\n   PERFECT — All tests passed!")
    elif pct >= 80:
        print("\n   GOOD — Agent is mostly working")
    elif pct >= 50:
        print("\n    PARTIAL — Several things need fixing")
    else:
        print("\n   CRITICAL — Major issues found")

    # Save report
    report_path = "test_report_indepth.txt"
    with open(report_path, "w") as f:
        f.write("DESKTOP AGENT IN-DEPTH TEST REPORT\n")
        f.write(f"Date : {time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write(f"Score: {len(passed)}/{total} ({pct}%)\n\n")
        for name, status in results:
            f.write(f"{status}  {name}\n")
        if failed:
            f.write(f"\nFAILED:\n")
            for name, _ in failed:
                f.write(f"  - {name}\n")

    print(f"\n   Full report saved: {report_path}\n")
